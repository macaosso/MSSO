核心演算法與程式碼

系統完全基於乾淨、模組化的原生 JavaScript 函式實作，便於輕鬆整合至您的前端網頁、行動App或後端系統。

rainfall_matrix.js 
        // 警告矩陣降雨影響評估函式 (結合 降雨機率 與 降雨量)
        function getRainfallImpactLevel(probability, precipitationSum) {
            if (precipitationSum <= 1.0) {
                return { label: '低', class: 'bg-blue-100 text-blue-700 dark:bg-blue-950 dark:text-blue-300' };
            }

            let impactScore = 0;
            if (precipitationSum >= 10) impactScore = 4;
            else if (precipitationSum >= 8) impactScore = 3;
            else if (precipitationSum >= 6) impactScore = 2;
            else if (precipitationSum >= 4) impactScore = 1;

            let likelihoodScore = 0;
            if (probability >= 90) likelihoodScore = 4;
            else if (probability >= 70) likelihoodScore = 3;
            else if (probability >= 50) likelihoodScore = 2;
            else if (probability >= 30) likelihoodScore = 1;

            const matrixValue = impactScore * likelihoodScore;

            if (matrixValue >= 16 || (impactScore >= 4 && likelihoodScore >= 4)) {
                return { label: '極高', class: 'bg-purple-100 text-purple-700 dark:bg-purple-950 dark:text-purple-300 font-bold' };
            } else if (matrixValue >= 8 || (impactScore >= 3)) {
                return { label: '高', class: 'bg-red-100 text-red-700 dark:bg-red-950 dark:text-red-300 font-bold' };
            } else if (matrixValue >= 3 || (impactScore >= 3)) {
                return { label: '中', class: 'bg-amber-100 text-amber-800 dark:bg-amber-950 dark:text-amber-300 font-semibold' };
            } else {
                return { label: '低', class: 'bg-blue-100 text-blue-700 dark:bg-blue-950 dark:text-blue-300' };
            }
        }

        // 等級標籤樣式：順序為 低 | 中 | 高 | 極高，縮小字體並套用對應背景色
        function getLevelStyle(level) {
            switch (level) {
                case '低':
                    return 'bg-slate-200 text-slate-700 dark:bg-slate-800 dark:text-slate-300 text-xs font-medium px-2 py-0.5 rounded';
                case '中':
                    return 'bg-yellow-100 text-yellow-800 dark:bg-yellow-950 dark:text-yellow-300 text-xs font-semibold px-2 py-0.5 rounded';
                case '高':
                    return 'bg-orange-100 text-orange-800 dark:bg-orange-950 dark:text-orange-300 text-xs font-semibold px-2 py-0.5 rounded';
                case '極高':
                    return 'bg-red-100 text-red-800 dark:bg-red-950 dark:text-red-300 text-xs font-bold px-2 py-0.5 rounded';
                default:
                    return 'bg-slate-100 text-slate-600 dark:bg-slate-800 dark:text-slate-400 text-xs px-2 py-0.5 rounded';
            }
        }
